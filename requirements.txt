COMPARE_QUESTION = "What training techniques does Mistral 7B use to improve efficiency?"

strategies = {
    "MMR":                    chatbot_mmr,
    "Semantic Similarity":    chatbot_sim,
    "Hybrid (BM25+Semantic)": chatbot_hybrid,
}

for label, bot in strategies.items():
    # Use a fresh bot instance per strategy so history doesn't bleed over
    if   label == "MMR":
        r = get_retriever(vector_store, strategy="mmr")
    elif label == "Semantic Similarity":
        r = get_retriever(vector_store, strategy="similarity")
    else:
        r = get_retriever(vector_store, strategy="hybrid", chunks=chunks)

    fresh_bot = PaperMindChatbot(r)
    ans, srcs = fresh_bot.chat(COMPARE_QUESTION)
    show_response(COMPARE_QUESTION, ans, srcs, strategy=label)
